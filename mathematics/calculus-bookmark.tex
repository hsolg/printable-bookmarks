\section*{Quadratic Equations}

Given

\[
ax^2 + bx + c = 0
\]

The roots are

\[
x=
\frac{-b\pm\sqrt{b^2-4ac}}
{2a}
\]

The discriminant is

\[
\Delta=b^2-4ac
\]

\begin{tabular}{ll}
$\Delta>0$ & two real roots\\
$\Delta=0$ & one repeated root\\
$\Delta<0$ & two complex roots
\end{tabular}

\section*{Factoring}

Difference of squares

\[
a^2-b^2=(a-b)(a+b)
\]

Perfect squares

\[
(a+b)^2=a^2+2ab+b^2
\]

\[
(a-b)^2=a^2-2ab+b^2
\]

\vspace{4mm}
\hrule
\vspace{2mm}

\section*{Exponent Rules}

\[
a^m a^n=a^{m+n}
\]

\[
\frac{a^m}{a^n}=a^{m-n}
\]

\[
(a^m)^n=a^{mn}
\]

\[
(ab)^n=a^n b^n
\]

\[
a^{-n}=\frac1{a^n}
\]

\vspace{4mm}
\hrule
\vspace{2mm}

\section*{Logarithms}

\[
\log(ab)=\log a+\log b
\]

\[
\log\left(\frac ab\right)
=
\log a-\log b
\]

\[
\log(a^n)=n\log a
\]
